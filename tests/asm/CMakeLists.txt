function(assemble_image name source)
    add_custom_command(OUTPUT "${image_directory}/${name}.bin"
        BYPRODUCTS "${image_directory}/${name}.p" "${image_directory}/${name}.lst"
        COMMAND "${CMAKE_COMMAND}" "-DASL=${asl_program}" "-DP2BIN=${p2bin_program}"
            "-DSOURCE=${source}" "-DOUTPUT=${image_directory}/${name}"
            "-DWORKING_DIRECTORY=${image_directory}" -P "${CMAKE_CURRENT_SOURCE_DIR}/assemble.cmake"
        DEPENDS mdcomp-asl-host "${source}" "${CMAKE_CURRENT_SOURCE_DIR}/assemble.cmake" ${ARGN} VERBATIM)
    set(asm_images ${asm_images} "${image_directory}/${name}.bin" PARENT_SCOPE)
endfunction()

function(decoder_image name codec architecture source entry original self_modifying)
    cmake_parse_arguments(IMAGE "JUMP_RETURN;HEADERLESS" "PAYLOAD_OFFSET;CONFIGURATION;STACK_SCRATCH" "DEFINES" ${ARGN})
    set(headerless false)
    if(IMAGE_HEADERLESS)
        set(headerless true)
    endif()
    if(NOT IMAGE_STACK_SCRATCH)
        set(IMAGE_STACK_SCRATCH 0)
    endif()
    if(NOT IMAGE_CONFIGURATION)
        set(IMAGE_CONFIGURATION "default")
    endif()
    if(NOT IMAGE_PAYLOAD_OFFSET)
        set(IMAGE_PAYLOAD_OFFSET 0)
    endif()
    set(definitions "")
    foreach(definition IN LISTS IMAGE_DEFINES)
        if(NOT definition MATCHES "^([A-Za-z_][A-Za-z0-9_]*)=([0-9]+)$")
            message(FATAL_ERROR "Invalid decoder definition: ${definition}")
        endif()
        string(APPEND definitions "${CMAKE_MATCH_1} = ${CMAKE_MATCH_2}\n")
    endforeach()
    if(codec STREQUAL "snkrle")
        set(read_ahead 1)
    elseif(codec STREQUAL "enigma" OR codec STREQUAL "nemesis")
        set(read_ahead 2)
    else()
        set(read_ahead 0)
    endif()
    set(include_source "${source}")
    if(IS_ABSOLUTE "${source}")
        configure_file("${source}" "${image_directory}/${name}.source.asm" COPYONLY)
        set(include_source "${name}.source.asm")
    endif()
    if(architecture STREQUAL "m68000")
        math(EXPR payload "0x20000 + ${IMAGE_PAYLOAD_OFFSET}" OUTPUT_FORMAT HEXADECIMAL)
        string(REPLACE "0x" "$" payload "${payload}")
        set(image_entry "${entry}")
        if(IMAGE_JUMP_RETURN)
            set(image_entry DriverEntry)
            string(APPEND definitions "\treg comperx_return_reg,a5\nDriverEntry:\n\tlea DriverDone(pc),a5\n\tbra.w ${entry}\n")
        endif()
        set(header "\tcpu 68000\n\torg $1000\n\tdc.b \"MD68\"\n\tdc.l ${image_entry},${payload},$100\nv_ngfx_buffer equ $FFFF8000\nvdp_data_port equ $C00000\n")
    else()
        set(header "\tcpu Z80\n\torg 1000h\n\tdb \"MDZ8\"\n\tdw ${entry},4000h,0100h\nOptimiseDriver equ 0\nzMusicData equ 8000h\n")
        if(self_modifying)
            string(APPEND header "\tdw zGetNextByte+1,zDecEndOrGetByte+1\n")
        endif()
    endif()
    set(wrapper "${image_directory}/${name}.driver.asm")
    set(footer "")
    if(IMAGE_JUMP_RETURN)
        set(footer "DriverDone:\n\trts\n")
    endif()
    file(CONFIGURE OUTPUT "${wrapper}" CONTENT "${header}${definitions}\tinclude \"${include_source}\"\n${footer}" @ONLY)
    set(dependencies ${decoder_sources})
    if(IS_ABSOLUTE "${source}")
        list(APPEND dependencies "${source}")
    endif()
    assemble_image(${name} "${wrapper}" ${dependencies})
    set(asm_images ${asm_images} PARENT_SCOPE)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/decoder_images.inc" "{\"${name}\", \"${codec}\", asm_test::cpu::${architecture}, ${original}, ${self_modifying}, ${read_ahead}, \"${IMAGE_CONFIGURATION}\", ${IMAGE_STACK_SCRATCH}, ${headerless}},\n")
    list(APPEND decoder_entries "{\"name\":\"${name}\",\"codec\":\"${codec}\",\"architecture\":\"${architecture}\",\"original_derived\":${original},\"self_modifying\":${self_modifying},\"readable_padding\":${read_ahead},\"source\":\"${include_source}\",\"configuration\":\"${IMAGE_CONFIGURATION}\",\"stack_scratch_bytes\":${IMAGE_STACK_SCRATCH},\"headerless\":${headerless}}")
    set(decoder_entries "${decoder_entries}" PARENT_SCOPE)
endfunction()

    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../references/sega-decoders.json" sega_manifest)

    math(EXPR source_last "${source_count} - 1")

        string(JSON filename GET "${sega_manifest}" sources ${index} name)

        endif()

        if(NOT actual_hash STREQUAL expected_hash)
            message(FATAL_ERROR "Pinned Sega source hash mismatch: ${filename}; rerun the explicit importer.")
        endif()
    endforeach()
    foreach(pair kosinski enigma nemesis saxman-z80)
        if(NOT EXISTS "${MDCOMP_SEGA_DECODER_DIR}/${pair}.asm")
            message(FATAL_ERROR "Missing imported Sega source ${pair}.asm; run tests/support/import_sega_decoders.py.")
        endif()
    endforeach()
