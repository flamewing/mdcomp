cmake_minimum_required(VERSION 3.30)
project(mdcomp_installed_smoke LANGUAGES CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(mdcomp 0.2 CONFIG REQUIRED PATHS "${PACKAGE_CONFIG_DIR}" NO_DEFAULT_PATH)
enable_testing()
foreach(linkage IN ITEMS static shared)
    set(suffix "")
    if(linkage STREQUAL "static")
        set(suffix Static)
    endif()
    add_executable(installed-${linkage} main.cc)
    foreach(codec IN ITEMS comper comperx enigma kosinski kosplus lzkn1 nemesis rocket saxman snkrle)
        target_link_libraries(installed-${linkage} PRIVATE mdcomp::${codec}${suffix})
        if(MSVC AND linkage STREQUAL "shared")
            string(TOUPPER "${CMAKE_BUILD_TYPE}" config)
            get_target_property(import mdcomp::${codec} IMPORTED_IMPLIB_${config})
            get_target_property(archive mdcomp::${codec}Static IMPORTED_LOCATION_${config})
            if(NOT import OR NOT archive)
                message(FATAL_ERROR "${codec}: missing installed ${config} library metadata")
            endif()
            if(import STREQUAL archive)
                message(FATAL_ERROR "${codec}: static archive collides with DLL import library: ${import}")
            endif()
        endif()
    endforeach()
    add_test(NAME installed-${linkage} COMMAND installed-${linkage})
endforeach()
if(WIN32)
    set(dll_names "constexpr wchar_t const* codec_dlls[] = {\n")
    foreach(codec IN ITEMS comper comperx enigma kosinski kosplus lzkn1 nemesis rocket saxman snkrle)
        string(APPEND dll_names "L\"$<TARGET_FILE_NAME:mdcomp::${codec}>\",\n")
    endforeach()
    string(APPEND dll_names "};\n")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/codec_dlls.hh" CONTENT "${dll_names}")
    target_include_directories(installed-shared PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
    target_compile_definitions(installed-shared PRIVATE CHECK_LOADED_DLLS)
    set_tests_properties(installed-shared PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:mdcomp::comper>")
endif()
